ALGORITHM MinDistance(A[0..n − 1])
    //Input: Array A[0..n − 1] of numbers
    //Output: Minimum distance between any two elements of the array
    
    sort A in ascending order

    dmin ← |A[1] − A[0]|

    for i ← 1 to n − 2 do
        d ← |A[i + 1] − A[i]|

        if d < dmin
            dmin ← d

    return dmin
