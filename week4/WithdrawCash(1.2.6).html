ALGORITHM WithdrawCash(card, pin, amount)
    //Input: Card, PIN, and withdrawal amount
    //Output: Cash withdrawal result

    if card is invalid
        return "Invalid card"

    if pin is incorrect
        return "Incorrect PIN"

    if amount ≤ 0
        return "Invalid amount"

    if amount > account_balance
        return "Insufficient funds"

    if amount > atm_cash
        return "ATM has insufficient cash"

    if amount cannot be dispensed
        return "Invalid amount"

    account_balance ← account_balance - amount
    atm_cash ← atm_cash - amount

    dispense amount in banknotes

    return "Cash dispensed successfully"
