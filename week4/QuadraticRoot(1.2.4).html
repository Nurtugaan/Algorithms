ALGORITHM FindRoots(a, b, c)

    if a = 0
        if b = 0
            if c = 0
                return "All real numbers"
            else
                return "No real roots"
        else
            x ← -c / b
            return x

    D ← b * b - 4 * a * c

    if D < 0
        return "No real roots"

    else if D = 0
        x ← -b / (2 * a)
        return x

    else
        x1 ← (-b + sqrt(D)) / (2 * a)
        x2 ← (-b - sqrt(D)) / (2 * a)
        return x1, x2
